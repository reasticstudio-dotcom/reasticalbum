function buatLinkAlbum() { const clientName = document.getElementById('client-name').value.trim(); const maxPhotos = document.getElementById('max-photos').value.trim(); if (!clientName || !maxPhotos) { alert('Nama klien dan batas maksimal foto harus diisi!'); return; } // Buat slug dari nama klien (contoh: "Bayu" jadi "bayu") const slug = clientName.toLowerCase().replace(/[^a-z0-9]/g, '-'); // Mengambil origin (contoh: https://reasticstudio-dotcom.github.io) let origin = window.location.origin; // Mengambil path repository (contoh: /reasticalbum/admin.html -> /reasticalbum/) let pathArray = window.location.pathname.split('/'); let repoName = pathArray[1] ? '/' + pathArray[1] + '/' : '/'; // Gabungkan menjadi link yang pasti benar menuju index.html const uniqueUrl = `${origin}${repoName}index.html?album=${slug}&max=${maxPhotos}`; // Tampilkan hasil box document.getElementById('generated-url').value = uniqueUrl; document.getElementById('result-box').classList.remove('hidden'); // Tambahkan ke tabel riwayat tambahkanKeRiwayat(clientName, maxPhotos, uniqueUrl); }
